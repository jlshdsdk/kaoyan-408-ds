直接插入排序代码(C/C++实现)

直接插入：移动腾位后写入 key

方向键也可切换

要点

  • 外层从下标 1 起，依次把 a[i] 插入左侧
  • 内层 while 把大于 key 的元素右移一格
  • 相等不右移，相对次序不变，故稳定
  • 打印每趟后数组，便于对照动画

时间最好 O(n)、最坏/平均 O(n^2)；空间 O(1)；稳定

完整 C++ 代码

// 直接插入排序代码（不带哨兵），打印每趟
#include <iostream>
using namespace std;

void print_arr(const int a[], int n) {
    for (int i = 0; i < n; i++) {
        if (i) cout << " ";
        cout << a[i];
    }
    cout << "\n";
}

void insert_sort(int a[], int n) {
    if (n <= 0) {
        cout << "error\n";
        return;
    }
    cout << "init: ";
    print_arr(a, n);
    for (int i = 1; i < n; i++) {
        int key = a[i];
        int j = i - 1;
        while (j >= 0 && a[j] > key) {
            a[j + 1] = a[j];
            j--;
        }
        a[j + 1] = key;
        cout << "pass " << i << ": ";
        print_arr(a, n);
    }
}

int main() {
    int a[] = {5, 2, 8, 1, 9, 3};
    int n = 6;
    insert_sort(a, n);
    cout << "result: ";
    print_arr(a, n);
    return 0;
}
