简单选择排序

每趟从未排序段选出最小元换到前端

方向键也可切换

要点

  • 第 i 趟在 a[i..n-1] 中找最小值下标
  • 与 a[i] 交换后，前 i+1 个已就位
  • 比较次数固定，与初始顺序关系不大
  • 不稳定：相等元可能因交换改变相对次序

时间最好/最坏/平均均 O(n^2)；空间 O(1)；不稳定

完整 C++ 代码

下载 .cpp 

// 简单选择排序思路：每趟选最小元换到前面
#include <iostream>
using namespace std;

void printArr(const int a[], int n) {
    for (int i = 0; i < n; i++) {
        if (i) cout << " ";
        cout << a[i];
    }
    cout << "\n";
}

void selectSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int k = i;
        for (int j = i + 1; j < n; j++) {
            if (a[j] < a[k]) k = j;
        }
        if (k != i) {
            int t = a[i];
            a[i] = a[k];
            a[k] = t;
        }
    }
}

int main() {
    int a[] = {5, 2, 8, 1, 9, 3};
    int n = 6;
    printArr(a, n);
    selectSort(a, n);
    printArr(a, n);
    return 0;
}
