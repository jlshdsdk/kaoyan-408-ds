归并排序代码(C/C++递归实现)

递归拆半再归并，每次合并后打印

方向键也可切换

要点

  • 先递归左半、再递归右半，最后合并
  • 合并用辅助数组，a[i]<=a[j] 取左保证稳定
  • 每次 merge 后打印当前整表状态
  • 空区间与单元素区间直接返回

时间最好/最坏/平均 O(n log n)；空间 O(n)；稳定

// 归并排序代码（递归）：打印每一层归并结果
#include <iostream>
using namespace std;

void printArr(const int a[], int n) {
    for (int i = 0; i < n; i++) {
        if (i) cout << " ";
        cout << a[i];
    }
    cout << "\n";
}

void merge(int a[], int L, int M, int R, int tmp[]) {
    int i = L, j = M + 1, k = L;
    while (i <= M && j <= R) {
        if (a[i] <= a[j]) tmp[k++] = a[i++];
        else tmp[k++] = a[j++];
    }
    while (i <= M) tmp[k++] = a[i++];
    while (j <= R) tmp[k++] = a[j++];
    for (int t = L; t <= R; t++) a[t] = tmp[t];
}

void mergeSort(int a[], int L, int R, int tmp[], int n) {
    if (L >= R) return;
    int M = L + (R - L) / 2;
    mergeSort(a, L, M, tmp, n);
    mergeSort(a, M + 1, R, tmp, n);
    merge(a, L, M, R, tmp);
    printArr(a, n);  // 每次归并后打印整表
}

int main() {
    int a[] = {5, 2, 8, 1, 9, 3};
    int n = 6;
    int tmp[6];
    printArr(a, n);
    mergeSort(a, 0, n - 1, tmp, n);
    return 0;
}
