冒泡排序

相邻比较交换，大的沉到末尾

方向键也可切换

要点

  • 一趟从左扫到右，相邻逆序就交换
  • 每趟结束后，当前最大沉到无序区末尾
  • 相等不交换，相对次序不变，故稳定
  • 共需多趟，直到有序区扩满全表

时间最好/最坏/平均均 O(n^2)（无提前结束）；空间 O(1)；稳定

// 冒泡排序：一趟把当前最大沉到末尾
#include <iostream>
using namespace std;

void print_pass(const int a[], int n, int pass) {
    cout << "pass " << pass << ":";
    for (int i = 0; i < n; i++) cout << " " << a[i];
    cout << "\n";
}

void bubble_sort(int a[], int n) {
    if (n <= 0) {
        cout << "error\n";
        return;
    }
    print_pass(a, n, 0);
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                int t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
            }
        }
        print_pass(a, n, i + 1);
    }
}

int main() {
    int a[] = {5, 2, 8, 1, 9, 3};
    int n = 6;
    bubble_sort(a, n);
    cout << "result:";
    for (int i = 0; i < n; i++) cout << " " << a[i];
    cout << "\n";
    return 0;
}
