折半查找(二分查找)

有序表取中点，一半一半缩小区间

方向键也可切换

要点

  • 前提是升序（或降序）静态表
  • 每轮看 mid，决定去左半还是右半
  • 打印 low/mid/high 方便对照过程
  • 失败时 low>high，区间被夹空

时间最好O(1)、最坏/平均O(log n)；空间迭代O(1)、递归O(log n)

完整 C++ 代码

下载 .cpp 

// 折半查找：有序表上比较，打印 low mid high
#include <iostream>
using namespace std;

// 返回下标；找不到返回 -1。过程打印 low/mid/high
int binarySearch(const int a[], int n, int key) {
    if (n <= 0) return -1;
    int low = 0, high = n - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2;
        cout << "low=" << low << " mid=" << mid << " high=" << high
             << " a[mid]=" << a[mid] << "\n";
        if (a[mid] == key) return mid;
        if (a[mid] < key) low = mid + 1;
        else high = mid - 1;
    }
    return -1;
}

int main() {
    int a[] = {2, 5, 8, 12, 16, 23, 38};
    int n = 7;

    cout << "search 16\n";
    int p = binarySearch(a, n, 16);
    if (p >= 0) cout << "found index=" << p << "\n";
    else cout << "error\n";

    cout << "search 7\n";
    p = binarySearch(a, n, 7);
    if (p >= 0) cout << "found index=" << p << "\n";
    else cout << "not found\n";

    // 空表
    p = binarySearch(a, 0, 1);
    if (p < 0) cout << "empty=-1\n";
    return 0;
}
