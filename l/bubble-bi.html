双向冒泡排序代码(C/C++实现)

双向冒泡：左右各沉浮一轮

方向键也可切换

要点

  • 一趟先从左到右，把当前最大沉到右端
  • 再从右到左，把当前最小浮到左端
  • 两端同时收缩有序区，趟数常更少
  • 相等不交换，仍是稳定排序

时间最好 O(n)、最坏/平均 O(n^2)；空间 O(1)；稳定

完整 C++ 代码

// 双向冒泡排序：一趟先左到右再右到左
#include <iostream>
using namespace std;

void print_arr(const int a[], int n) {
    for (int i = 0; i < n; i++) {
        if (i) cout << " ";
        cout << a[i];
    }
    cout << "\n";
}

void bubble_bi_sort(int a[], int n) {
    if (n <= 0) {
        cout << "error\n";
        return;
    }
    cout << "init: ";
    print_arr(a, n);
    int left = 0, right = n - 1;
    int pass = 0;
    while (left < right) {
        pass++;
        for (int i = left; i < right; i++) {
            if (a[i] > a[i + 1]) {
                int t = a[i];
                a[i] = a[i + 1];
                a[i + 1] = t;
            }
        }
        right--;
        cout << "pass " << pass << " L->R: ";
        print_arr(a, n);

        for (int i = right; i > left; i--) {
            if (a[i - 1] > a[i]) {
                int t = a[i - 1];
                a[i - 1] = a[i];
                a[i] = t;
            }
        }
        left++;
        cout << "pass " << pass << " R->L: ";
        print_arr(a, n);
    }
}

int main() {
    int a[] = {5, 2, 8, 1, 9, 3};
    int n = 6;
    bubble_bi_sort(a, n);
    cout << "result: ";
    print_arr(a, n);
    return 0;
}
