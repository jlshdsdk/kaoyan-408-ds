希尔排序

按增量分组，组内插入再缩小

方向键也可切换

要点

  • n=6 时先 gap=3 再 gap=1，组内像插入
  • 同组元素相隔 gap，先粗调再细调
  • hl 标正在比较的两个下标，note 写分组
  • 远距交换可打乱相等元次序，不稳定

时间依赖增量，常见约 O(n^1.3) 到 O(n^2)；空间 O(1)；不稳定

// 希尔排序：增量 3 再 1（n=6），组内插入
#include <iostream>
using namespace std;

void print_arr(const int a[], int n, const char* tag) {
    cout << tag;
    for (int i = 0; i < n; i++) cout << " " << a[i];
    cout << "\n";
}

void shell_sort(int a[], int n) {
    if (n <= 0) {
        cout << "error\n";
        return;
    }
    print_arr(a, n, "init:");
    // n=6 时增量序列固定为 3,1
    int gaps[] = {3, 1};
    for (int g = 0; g < 2; g++) {
        int gap = gaps[g];
        cout << "gap=" << gap << "\n";
        for (int i = gap; i < n; i++) {
            int key = a[i];
            int j = i - gap;
            cout << "  cmp idx " << j << " and " << i << "\n";
            while (j >= 0 && a[j] > key) {
                a[j + gap] = a[j];
                j -= gap;
            }
            a[j + gap] = key;
        }
        print_arr(a, n, "after:");
    }
}

int main() {
    int a[] = {5, 2, 8, 1, 9, 3};
    int n = 6;
    shell_sort(a, n);
    print_arr(a, n, "result:");
    return 0;
}
