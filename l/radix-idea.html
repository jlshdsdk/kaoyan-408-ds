基数排序

LSD 按位分配进桶，再按序收集

方向键也可切换

要点

  • 最低位优先：先个位桶，再十位桶
  • 同桶内保持原相对次序，故整体稳定
  • 分配是入桶，收集是按桶号 0..9 接回
  • 位数固定时时间近似线性

时间 O(d(n+r))，d 位数 r 基数；空间 O(n+r)；稳定

// 基数排序思路：LSD，先按个位再按十位分配收集
#include <iostream>
using namespace std;

void printArr(const int a[], int n) {
    for (int i = 0; i < n; i++) {
        if (i) cout << " ";
        cout << a[i];
    }
    cout << "\n";
}

// 按某十进制位（exp=1 个位，exp=10 十位）分配到 10 个桶再收集
void countPass(int a[], int n, int exp) {
    int bucket[10][8];
    int cnt[10] = {0};
    for (int i = 0; i < n; i++) {
        int d = (a[i] / exp) % 10;
        bucket[d][cnt[d]++] = a[i];
    }
    int k = 0;
    for (int d = 0; d < 10; d++) {
        for (int j = 0; j < cnt[d]; j++) a[k++] = bucket[d][j];
    }
}

void radixSort(int a[], int n) {
    countPass(a, n, 1);   // 个位
    printArr(a, n);
    countPass(a, n, 10);  // 十位
    printArr(a, n);
}

int main() {
    int a[] = {21, 15, 8, 32, 4, 27};
    int n = 6;
    printArr(a, n);
    radixSort(a, n);
    return 0;
}
