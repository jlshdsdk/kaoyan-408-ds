图的遍历-深度优先搜索(DFS)代码(C/C++实现)

递归 DFS：进入标访问，扫邻居，离开即出栈

方向键也可切换

要点

  • enter 对应入栈，leave 对应出栈返回
  • visit 顺序仍是 A B D C E
  • 已访问邻居打印 skip，避免重复深入
  • 需 vis 数组，防止有向环导致死递归

时间 O(n+m)；递归栈最坏 O(n)；额外 vis 空间 O(n)

完整 C++ 代码

下载 .cpp 

// 有向图 DFS 完整递归：打印入栈/出栈与访问序
#include <iostream>
#include <vector>
using namespace std;

const int N = 5;
const char* name[N] = {"A", "B", "C", "D", "E"};
bool vis[N];

void dfs(int u, vector<int> adj[N]) {
    vis[u] = true;
    cout << "enter " << name[u] << "\n";
    cout << "visit " << name[u] << "\n";
    for (int v : adj[u]) {
        if (!vis[v]) {
            dfs(v, adj);
        } else {
            cout << "skip " << name[v] << "\n";
        }
    }
    cout << "leave " << name[u] << "\n";
}

int main() {
    vector<int> adj[N];
    adj[0] = {1, 2};  // A -> B, C
    adj[1] = {3};     // B -> D
    adj[2] = {3, 4};  // C -> D, E
    adj[3] = {};
    adj[4] = {};

    cout << "DFS from A\n";
    for (int i = 0; i < N; i++) vis[i] = false;
    dfs(0, adj);
    cout << "order: A B D C E\n";
    return 0;
}
